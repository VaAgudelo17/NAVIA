# --- FRAMEWORK WEB ---
# FastAPI: Framework moderno y rápido para crear APIs REST en Python.
fastapi>=0.115.0

# Uvicorn: Servidor ASGI de alto rendimiento para ejecutar aplicaciones FastAPI.
uvicorn[standard]>=0.32.0

# python-multipart: Necesario para manejar archivos subidos (form-data).
python-multipart>=0.0.12

# --- PROCESAMIENTO DE IMÁGENES ---
# OpenCV (cv2): Librería de visión por computadora más usada en el mundo.
# La versión "headless" no incluye interfaz gráfica (ideal para servidores).
opencv-python-headless>=4.10.0

# Pillow: Librería para manipulación de imágenes en Python.
Pillow>=11.0.0

# NumPy: Librería fundamental para computación numérica.
numpy>=2.0.0

# --- OCR (Reconocimiento Óptico de Caracteres) ---
# pytesseract: Wrapper de Python para Tesseract OCR.
pytesseract>=0.3.13

# --- PROCESAMIENTO DE PDF ---
# PyMuPDF (fitz): Librería rápida para extraer contenido de PDFs.
# Alternativa: pdf2image (más lenta, requiere poppler)
pymupdf>=1.23.0

# --- DETECCIÓN DE OBJETOS Y TRACKING ---
# Ultralytics: YOLO-World (detección) + ByteTrack (tracking).
ultralytics>=8.3.0
# lap: requerido por ByteTrack (tracking). Sin esto ultralytics lo instala
# en runtime causando un spike de memoria justo al cargar el modelo.
lap>=0.5.12

# --- WEBSOCKET ---
# websockets: Soporte WebSocket para detección en tiempo real.
websockets>=12.0

# --- MODELOS DE IA (Hugging Face) ---
# Transformers: Librería de Hugging Face para modelos de deep learning.
# Usamos: Depth Anything V2 (profundidad), Florence-2 (captioning),
# CLIP (filtrado semántico opcional).
transformers>=4.41.0

# --- FLORENCE-2 (CAPTIONING) ---
# Dependencias requeridas por Florence-2 para captioning de escenas.
einops>=0.7.0
timm>=0.9.0

# --- TTS (PIPER + gTTS) ---
# Piper TTS: síntesis de voz offline con modelos VITS (ONNX).
# Genera audio WAV en español para la app móvil.
piper-tts>=1.2.0

# gTTS: Google TTS fallback cuando Piper no está disponible.
gtts>=2.5.0

# --- GEMINI (GOOGLE AI) ---
# Google Generative AI SDK: OCR avanzado con Gemini Vision
# Gemini Flash: gratis, rapido, excelente para documentos
google-genai>=1.0.0

# --- OPENAI ---
# SDK oficial de OpenAI: GPT-4o Vision para validación y narrativa empática en modo exploración
openai>=1.0.0

# --- UTILIDADES ---
# Pydantic: Validación de datos y configuración.
pydantic>=2.9.0

# pydantic-settings: Extensión para manejar configuración con Pydantic v2
pydantic-settings>=2.6.0

# python-dotenv: Cargar variables de entorno desde archivo .env
python-dotenv>=1.0.1

# aiofiles: Manejo asíncrono de archivos.
aiofiles>=24.1.0

# --- BASE DE DATOS ---
# SQLAlchemy: ORM para modelado y consultas a la base de datos.
# Incluye soporte async para integración con FastAPI.
sqlalchemy[asyncio]>=2.0.0

# aiosqlite: Driver async para SQLite (desarrollo).
# En producción se puede cambiar por asyncpg (PostgreSQL).
aiosqlite>=0.20.0

# Alembic: Herramienta de migraciones para SQLAlchemy.
alembic>=1.13.0
